import json
import random
import sys
import time
from pathlib import Path
import numpy as np
import torch
from t2l.t2l import process
from t2l.mtl import utils

out = Path('/Users/xcy/Program/auto-lrc/reports/p0/original')
out.mkdir(parents=True, exist_ok=True)
torch.set_num_threads(2)
lyrics = Path('demofile/original_txt.txt').read_text().splitlines()
capture = {}
for function in ('alignment', 'alignment_bdr'):
    old = getattr(utils, function)
    def record(*args, _old=old, **kwargs):
        frames, score = _old(*args, **kwargs)
        capture.update(word_frames=frames, score=float(score), phonemes=list(args[1]),
                       word_phoneme_indices=np.asarray(args[2]).tolist(), output_frames=len(args[0]))
        return frames, score
    setattr(utils, function, record)
for method in ('MTL', 'MTL_BDR'):
    random.seed(7)
    np.random.seed(7)
    torch.manual_seed(7)
    start = time.perf_counter()
    lrc = process(lyrics, 'demofile/original_track.mp3', mtl_model=method, vocalize=False, verbose=True)
    capture.update(method=method, seconds=time.perf_counter()-start, lrc=lrc)
    (out/(method.lower()+'.json')).write_text(json.dumps(capture, ensure_ascii=False, indent=2), encoding='utf8')
    (out/(method.lower()+'.lrc')).write_text(lrc, encoding='utf8')
    print(method, capture['seconds'],len(capture['word_frames']), flush=True)
